cmake_minimum_required(VERSION 3.16)
project(LearnedIndexModel LANGUAGES C CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

include(CTest)
add_subdirectory(src)
if(BUILD_TESTING)
  add_executable(blocklearn_test tests/blocklearn_test.cpp)
  target_link_libraries(blocklearn_test PRIVATE blocklearn_adapter)
  add_test(NAME blocklearn_semantics COMMAND blocklearn_test)
  add_executable(baseline_contract_test tests/baseline_contract_test.cpp)
  target_link_libraries(baseline_contract_test PRIVATE art_baseline hot_baseline)
  add_test(NAME baseline_contract COMMAND baseline_contract_test)
  add_executable(model_certificate_test tests/model_certificate_test.cpp)
  target_link_libraries(model_certificate_test PRIVATE blocklearn_core)
  add_test(NAME model_certificates COMMAND model_certificate_test)
  find_package(Python3 COMPONENTS Interpreter QUIET)
  if(Python3_Interpreter_FOUND)
    add_test(NAME experiment_contract COMMAND "${Python3_EXECUTABLE}"
      "${CMAKE_CURRENT_SOURCE_DIR}/tests/experiment_contract_test.py" "$<TARGET_FILE:blocklearn_mixed>")
  endif()
endif()
