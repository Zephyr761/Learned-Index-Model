cmake_minimum_required(VERSION 3.16)
project(LearnedIndexModel LANGUAGES C CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

include(CTest)
add_subdirectory(src)
if(BUILD_TESTING)
  add_executable(blocklearn_test tests/blocklearn_test.cpp)
  target_link_libraries(blocklearn_test PRIVATE blocklearn_adapter)
  add_test(NAME blocklearn_semantics COMMAND blocklearn_test)
  add_executable(baseline_contract_test tests/baseline_contract_test.cpp)
  target_link_libraries(baseline_contract_test PRIVATE art_baseline hot_baseline)
  add_test(NAME baseline_contract COMMAND baseline_contract_test)
  get_target_property(BLOCKLEARN_TEST_NATIVE_HOT hot_baseline BLOCKLEARN_NATIVE_HOT)
  if(BLOCKLEARN_TEST_NATIVE_HOT)
    add_executable(hot_native_contract_test tests/hot_native_contract_test.cpp)
    target_link_libraries(hot_native_contract_test PRIVATE hot_baseline)
    add_test(NAME hot_native_contract COMMAND hot_native_contract_test)
  endif()
  add_executable(model_certificate_test tests/model_certificate_test.cpp)
  target_link_libraries(model_certificate_test PRIVATE blocklearn_core)
  add_test(NAME model_certificates COMMAND model_certificate_test)
  add_executable(lookup_profile_test tests/lookup_profile_test.cpp)
  target_link_libraries(lookup_profile_test PRIVATE blocklearn_core)
  add_test(NAME lookup_profile COMMAND lookup_profile_test)
  add_executable(query_optimizations_test tests/query_optimizations_test.cpp)
  target_link_libraries(query_optimizations_test PRIVATE blocklearn_adapter)
  add_test(NAME query_optimizations COMMAND query_optimizations_test)
  find_package(Python3 COMPONENTS Interpreter QUIET)
  if(Python3_Interpreter_FOUND)
    add_test(NAME experiment_contract COMMAND "${Python3_EXECUTABLE}"
      "${CMAKE_CURRENT_SOURCE_DIR}/tests/experiment_contract_test.py" "$<TARGET_FILE:blocklearn_mixed>")
    add_test(NAME profile_runner_contract COMMAND "${Python3_EXECUTABLE}"
      "${CMAKE_CURRENT_SOURCE_DIR}/tests/profile_runner_test.py" "$<TARGET_FILE:blocklearn_profile>")
    add_test(NAME ablation_contract COMMAND "${Python3_EXECUTABLE}"
      "${CMAKE_CURRENT_SOURCE_DIR}/tests/ablation_contract_test.py" "$<TARGET_FILE:blocklearn_mixed>")
    add_test(NAME linux_hot_entry_contract COMMAND "${Python3_EXECUTABLE}" -B
      "${CMAKE_CURRENT_SOURCE_DIR}/tests/linux_hot_prepare_test.py")
  endif()
endif()
