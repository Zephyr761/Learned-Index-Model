cmake_minimum_required(VERSION 3.16)
project(BlockLearn LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(ART_SOURCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../related_works/libart" CACHE PATH "Unmodified armon/libart checkout")
set(HOT_SOURCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../related_works/hot" CACHE PATH "Unmodified speedskater/hot checkout")
option(BLOCKLEARN_SANITIZERS "Address/undefined sanitizers (not for performance runs)" OFF)
option(BLOCKLEARN_REQUIRE_HOT "Fail configure unless native HOT is usable" OFF)
add_library(blocklearn_contract INTERFACE)
if(DEFINED STRING_INDEX_CONTRACT_INCLUDE_DIR)
  target_include_directories(blocklearn_contract INTERFACE "${STRING_INDEX_CONTRACT_INCLUDE_DIR}")
else()
  target_include_directories(blocklearn_contract INTERFACE "${CMAKE_CURRENT_SOURCE_DIR}/benchmark/contract")
endif()
target_include_directories(blocklearn_contract INTERFACE "${CMAKE_CURRENT_SOURCE_DIR}")
target_compile_features(blocklearn_contract INTERFACE cxx_std_17)
if(BLOCKLEARN_SANITIZERS)
  target_compile_options(blocklearn_contract INTERFACE -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(blocklearn_contract INTERFACE -fsanitize=address,undefined)
endif()
add_library(blocklearn_core STATIC block_model.cpp residual_tags.cpp query_cost.cpp region_planner.cpp learned_index.cpp)
target_link_libraries(blocklearn_core PUBLIC blocklearn_contract)
target_compile_options(blocklearn_core PRIVATE -Wall -Wextra -Wpedantic)
add_library(blocklearn_adapter STATIC benchmark/model_factories.cpp)
target_link_libraries(blocklearn_adapter PUBLIC blocklearn_core)
add_library(art_baseline STATIC benchmark/art_baseline.cpp)
target_link_libraries(art_baseline PUBLIC blocklearn_contract)
if(EXISTS "${ART_SOURCE_DIR}/src/art.c")
  add_library(blocklearn_libart STATIC "${ART_SOURCE_DIR}/src/art.c")
  target_include_directories(blocklearn_libart PUBLIC "${ART_SOURCE_DIR}/src")
  target_compile_definitions(art_baseline PRIVATE BLOCKLEARN_HAVE_ART=1)
  target_link_libraries(art_baseline PRIVATE blocklearn_libart)
  if(BLOCKLEARN_SANITIZERS)
    target_compile_options(blocklearn_libart PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
  endif()
else()
  message(WARNING "ART source unavailable: baseline reports unsupported")
endif()
add_library(hot_baseline STATIC benchmark/hot_baseline.cpp)
target_link_libraries(hot_baseline PUBLIC blocklearn_contract)
set(BLOCKLEARN_HOT_USABLE OFF)
if(CMAKE_SYSTEM_PROCESSOR MATCHES "^(x86_64|AMD64|amd64)$" AND
   CMAKE_CXX_COMPILER_ID STREQUAL "GNU" AND
   EXISTS "${HOT_SOURCE_DIR}/libs/hot/single-threaded/include/hot/singlethreaded/HOTSingleThreaded.hpp" AND
   NOT CMAKE_CROSSCOMPILING)
  include(CheckCXXSourceRuns)
  check_cxx_source_runs("int main() { __builtin_cpu_init(); return !(__builtin_cpu_supports(\"avx2\") && __builtin_cpu_supports(\"bmi\") && __builtin_cpu_supports(\"bmi2\") && __builtin_cpu_supports(\"lzcnt\") && __builtin_cpu_supports(\"popcnt\")); }" BLOCKLEARN_CPU_HOT)
  if(BLOCKLEARN_CPU_HOT)
    set(BLOCKLEARN_HOT_USABLE ON)
    target_compile_definitions(hot_baseline PRIVATE BLOCKLEARN_NATIVE_HOT=1)
    target_compile_options(hot_baseline PRIVATE -mavx2 -mbmi -mbmi2 -mlzcnt -mpopcnt -Wno-deprecated-declarations)
    target_compile_options(blocklearn_core PRIVATE -mavx2)
    target_include_directories(hot_baseline SYSTEM PRIVATE
      "${HOT_SOURCE_DIR}/libs/hot/single-threaded/include"
      "${HOT_SOURCE_DIR}/libs/hot/commons/include"
      "${HOT_SOURCE_DIR}/libs/idx/content-helpers/include")
  endif()
endif()
if(NOT BLOCKLEARN_HOT_USABLE)
  if(BLOCKLEARN_REQUIRE_HOT)
    message(FATAL_ERROR "Native HOT requires GCC, x86-64 AVX2/BMI/BMI2/LZCNT/POPCNT and HOT_SOURCE_DIR")
  endif()
  message(STATUS "Native HOT unavailable; adapter explicitly reports unsupported (no emulation)")
endif()
set_property(TARGET hot_baseline PROPERTY BLOCKLEARN_NATIVE_HOT "${BLOCKLEARN_HOT_USABLE}")
add_executable(blocklearn_mixed benchmark/mixed_runner.cpp)
target_link_libraries(blocklearn_mixed PRIVATE blocklearn_adapter art_baseline hot_baseline)
add_executable(blocklearn_profile benchmark/profile_runner.cpp)
target_link_libraries(blocklearn_profile PRIVATE blocklearn_adapter)
add_executable(blocklearn_calibrate benchmark/calibrate_planner.cpp)
target_link_libraries(blocklearn_calibrate PRIVATE blocklearn_core)
if(BLOCKLEARN_HOT_USABLE)
  # tag_probe is inline: match the actual core's SIMD backend in calibration.
  target_compile_options(blocklearn_calibrate PRIVATE -mavx2)
endif()
